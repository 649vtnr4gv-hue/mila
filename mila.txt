  function maybeGreet() {
    if (S.greeted || !S.profile.setupDone || !S.profile.apiKey) return;
    S.greeted = true;
    const last = S.messages[S.messages.length - 1];
    if (!last) return askHer("[Служебная пометка, не его слова: он впервые открыл этот чат. Напиши ему первое сообщение: познакомься, коротко и тепло.]");
    const gap = Date.now() - last.ts;
    if (gap >= GREET_AFTER) askHer(`[Служебная пометка, не его слова: он только что открыл чат. Вы не общались ${ago(gap)}. Напиши ему первой: коротко и по-живому, можно вспомнить что-то из прошлого разговора.]`);
  }
  function memoryTick() {
    S.profile.sinceMemory = (S.profile.sinceMemory || 0) + 1;
    if (S.profile.sinceMemory >= 8 && !S.memBusy) updateMemory();
  }
  async function updateMemory() {
    S.memBusy = true;
    const p = S.profile;
    const lines = S.messages.slice(-24).map(m => (m.r === "u" ? "Он: " : p.herName + ": ") + m.t).join("\n");
    const prompt = `Уже известные факты о нём:
${p.facts.length ? p.facts.map(f => "- " + f).join("\n") : "нет"}
Прежнее описание разговоров: ${p.summary || "нет"}

Последние сообщения:
${lines}`;
    const sys = `Ты ведёшь память ИИ-собеседницы по имени ${p.herName} о человеке, с которым она переписывается. Обнови память. Факты — только то, что он сам сказал о себе и своей жизни: имя, работа, увлечения, планы, важные люди, что любит и не любит. Коротко, по-русски, без повторов, не больше 40 самых важных. Описание — 2–4 предложения о том, о чём они говорили недавно и в каком настроении. Верни JSON {"facts": ["..."], "summary": "..."}.`;
    try {
      const r = JSON.parse(await gemini(sys, [{ role: "user", content: prompt }], true));
      if (r && Array.isArray(r.facts)) p.facts = r.facts.map(clean).filter(Boolean).map(f => f.slice(0, 160)).slice(0, 40);
      if (r && typeof r.summary === "string") p.summary = clean(r.summary).slice(0, 900);
      p.sinceMemory = 0; persist();
    } catch (e) { /* в следующий раз */ }
    finally { S.memBusy = false; }
  }

  /* ---------- settings ---------- */
  let first = false;
  function renderFacts() {
    const ul = $("facts"); ul.textContent = "";
    S.profile.facts.forEach((f, i) => {
      const li = document.createElement("li"); const s = document.createElement("span"); s.textContent = f;
      const x = document.createElement("button"); x.type = "button"; x.className = "x"; x.textContent = "✕"; x.setAttribute("aria-label", "Забыть: " + f);
      x.onclick = () => { S.profile.facts.splice(i, 1); renderFacts(); persist(); };
      li.append(s, x); ul.append(li);
    });
    $("factsEmpty").hidden = S.profile.facts.length > 0;
  }
  function openSheet(isFirst) {
    first = !!isFirst;
    $("fName").value = S.profile.herName; $("fChar").value = S.profile.character; $("fAbout").value = S.profile.aboutMe;
    $("fKey").value = S.profile.apiKey; $("fModel").value = S.profile.model;
    $("sheetTitle").textContent = first ? "Какой она будет" : "Настройки";
    $("saveSettings").textContent = first ? "Начать" : "Сохранить";
    $("closeSettings").hidden = first; $("memBlock").hidden = first; $("dangerRule").hidden = first; $("dangerRow").hidden = first;
    renderFacts(); $("sheet").hidden = false;
  }
  $("openSettings").onclick = () => openSheet(false);
  $("closeSettings").onclick = () => { $("sheet").hidden = true; };
  $("saveSettings").onclick = () => {
    const p = S.profile;
    p.herName = clean($("fName").value) || DEFAULTS.herName;
    p.character = clean($("fChar").value) || DEFAULTS.character;
    p.aboutMe = clean($("fAbout").value);
    p.apiKey = clean($("fKey").value);
    p.model = clean($("fModel").value) || DEFAULTS.model;
    if (!p.apiKey) { $("keyHint").classList.add("warn"); $("fKey").focus(); return; }
    const wasFirst = !p.setupDone; p.setupDone = true;
    $("sheet").hidden = true; renderHeader(); persist(); $("notice").hidden = true;
    if (wasFirst) { renderLog(); maybeGreet(); }
  };
  let armTimer = null;
  $("wipe").onclick = (e) => {
    const b = e.currentTarget;
    if (!b.classList.contains("armed")) {
      b.classList.add("armed"); b.textContent = "Точно стереть? Нажми ещё раз";
      clearTimeout(armTimer); armTimer = setTimeout(() => { b.classList.remove("armed"); b.textContent = "Стереть переписку"; }, 4000);
      return;
    }
    clearTimeout(armTimer); b.classList.remove("armed"); b.textContent = "Стереть переписку";
    S.messages = []; S.profile.summary = ""; S.profile.sinceMemory = 0;
    persist(); renderLog(); $("sheet").hidden = true;
  };

  /* ---------- composer ---------- */
  const box = $("msg");
  function autosize() { box.style.height = "auto"; box.style.height = Math.min(box.scrollHeight, 140) + "px"; $("send").disabled = S.busy || !clean(box.value); }
  box.addEventListener("input", autosize);
  box.addEventListener("keydown", (e) => {
    if (e.key === "Enter" && !e.shiftKey && !e.isComposing && matchMedia("(pointer: fine)").matches) { e.preventDefault(); $("composer").requestSubmit(); }
  });
  $("composer").addEventListener("submit", sendMine);

  /* ---------- boot ---------- */
  load(); renderHeader(); renderLog(); autosize();
  if (!S.profile.setupDone || !S.profile.apiKey) openSheet(!S.profile.setupDone);
  else maybeGreet();
})();
